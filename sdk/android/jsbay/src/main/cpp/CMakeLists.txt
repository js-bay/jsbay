cmake_minimum_required(VERSION 3.22)
project(jsbay_android C)
if(NOT DEFINED JSBAY_JNI_LIBS)
  message(FATAL_ERROR "Set -DJSBAY_JNI_LIBS=<distribution>/jniLibs when building the Android SDK")
endif()
set(JSBAY_CORE "${JSBAY_JNI_LIBS}/${ANDROID_ABI}/libjsbay.so")
if(NOT EXISTS "${JSBAY_CORE}")
  message(FATAL_ERROR "Missing Android core library for ${ANDROID_ABI}: ${JSBAY_CORE}")
endif()
add_library(jsbay_core SHARED IMPORTED GLOBAL)
set_target_properties(jsbay_core PROPERTIES IMPORTED_LOCATION "${JSBAY_CORE}")
add_library(jsbay_android SHARED jsbay_android.c)
set(JSBAY_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../include" CACHE PATH "Directory containing jsbay.h")
if(NOT EXISTS "${JSBAY_INCLUDE_DIR}/jsbay.h")
  set(JSBAY_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../../include")
endif()
if(NOT EXISTS "${JSBAY_INCLUDE_DIR}/jsbay.h")
  message(FATAL_ERROR "Set JSBAY_INCLUDE_DIR to the directory containing jsbay.h")
endif()
target_include_directories(jsbay_android PRIVATE ${JSBAY_INCLUDE_DIR})
find_library(log-lib log)
target_link_libraries(jsbay_android PRIVATE jsbay_core ${log-lib})
