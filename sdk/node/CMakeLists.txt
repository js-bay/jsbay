cmake_minimum_required(VERSION 3.15)
project(jsbay_node LANGUAGES C CXX)

set(JSBAY_INCLUDE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../include" CACHE PATH "JsBay headers")
set(JSBAY_LIBRARY "" CACHE FILEPATH "JsBay static library")
set(NODE_INCLUDE "" CACHE PATH "Directory containing node_api.h")

if(NOT JSBAY_LIBRARY)
    message(FATAL_ERROR "Set JSBAY_LIBRARY to libjsbay.a")
endif()
if(NOT NODE_INCLUDE)
    message(FATAL_ERROR "Set NODE_INCLUDE to the target Node headers")
endif()

add_library(jsbay_node MODULE src/jsbay_node.cpp)
target_compile_features(jsbay_node PRIVATE cxx_std_17)
get_filename_component(NODE_INCLUDE_PARENT "${NODE_INCLUDE}" DIRECTORY)
target_include_directories(jsbay_node PRIVATE ${JSBAY_INCLUDE_DIR} ${NODE_INCLUDE} ${NODE_INCLUDE_PARENT})
target_link_libraries(jsbay_node PRIVATE ${JSBAY_LIBRARY})
if(APPLE)
    find_library(JAVASCRIPTCORE_FRAMEWORK JavaScriptCore REQUIRED)
    find_library(FOUNDATION_FRAMEWORK Foundation REQUIRED)
    target_link_options(jsbay_node PRIVATE "-undefined" "dynamic_lookup")
    target_link_libraries(jsbay_node PRIVATE ${JAVASCRIPTCORE_FRAMEWORK} ${FOUNDATION_FRAMEWORK})
elseif(UNIX)
    find_package(Threads REQUIRED)
    target_link_libraries(jsbay_node PRIVATE Threads::Threads ${CMAKE_DL_LIBS} m)
endif()
set_target_properties(jsbay_node PROPERTIES PREFIX "" SUFFIX ".node" LIBRARY_OUTPUT_DIRECTORY "$<1:${CMAKE_CURRENT_BINARY_DIR}>")
